Core Concepts
What is an Embedded System?
Embedded System

A computer built inside a device to do one fixed job. It is hidden — you don't see it as a computer.

  • ✓Single purpose — one job, does it forever
  • ✓Real-time — must respond within a fixed time
  • ✓Low power, small size, very low cost
  • ✓No OS or tiny RTOS only
  • e.g.Washing machine, pacemaker, traffic light, ATM, TV remote
General Purpose Computer

Designed to do any task the user chooses. Flexible, powerful, but expensive and large.

  • ✓Multi-purpose — runs any software
  • ✓Large RAM, big storage, full OS
  • ✓User interaction via keyboard, mouse, screen
  • ✗No guaranteed real-time response
  • e.g.Laptop, desktop, server
Real-Time Constraint

The most critical property of embedded systems.

TypeMiss deadline?Example
HardCatastrophic failurePacemaker, ABS brakes
SoftDegraded qualityVideo streaming
💡In ABS brakes, a 500ms late response = crash. Timing IS correctness.
CISC vs RISC — The Two Processor Philosophies
CISC — Complex Instruction Set Computer
  • IdeaGive the programmer powerful, complex instructions. Hardware does heavy work.
  • InstrLarge set — hundreds of instructions, variable length (1–15 bytes)
  • CyclesVariable — complex instructions take multiple clock cycles
  • MemoryAny instruction can access memory directly
  • RegsFew registers
  • e.g.Intel x86, 8051 — your laptop and 8051 kit
RISC — Reduced Instruction Set Computer
  • IdeaKeep instructions simple and uniform. Execute blazing fast. Compiler does the work.
  • InstrSmall set — ~100 instructions, fixed 32-bit size (in ARM)
  • CyclesMostly 1 cycle per instruction — predictable
  • MemoryLOAD/STORE only — arithmetic only between registers
  • RegsMany registers — ARM has 16
  • e.g.ARM (LPC2148) — your ARM board
Von Neumann vs Harvard Architecture
Von Neumann
CPU ←──── Single bus ────► One shared memory (code + data together) Problem: CPU can't fetch instruction AND data at the same time → BOTTLENECK
  • MemoryOne memory for both code and data
  • BusOne shared bus
  • SpeedSlower — can only do one at a time
  • e.g.Your PC / laptop
Harvard
CPU ←── Instruction bus ──► Program ROM ←── Data bus ──► Data RAM Both buses work SIMULTANEOUSLY → FASTER
  • MemorySeparate memory for code and data
  • BusTwo independent buses
  • SpeedFaster — parallel access
  • 8051Pure Harvard — ROM for code, RAM for data
  • ARMModified Harvard — separate caches, shared address space
Endianness — How Multi-byte Data Stored in Memory
Big Endian — MSB at lowest address
Store 0x12345678 at address 0x1000: Address 0x1000 → 0x12 ← MSB first Address 0x1001 → 0x34 Address 0x1002 → 0x56 Address 0x1003 → 0x78 ← LSB last Reads naturally left to right → 8051 is Big Endian
Little Endian — LSB at lowest address
Store 0x12345678 at address 0x1000: Address 0x1000 → 0x78 ← LSB first Address 0x1001 → 0x56 Address 0x1002 → 0x34 Address 0x1003 → 0x12 ← MSB last ARM default is Little Endian → LPC2148
Pipelining — ARM7 3-Stage Pipeline
How Pipelining Works
WITHOUT pipeline (sequential): Instr1: [Fetch]─[Decode]─[Execute] Instr2: [Fetch]─[Decode]─[Execute] → 6 time slots for 2 instructions WITH pipeline (ARM7 — 3 stages): Time: T1 T2 T3 T4 T5 I1: [Fetch][Decode][Execute] I2: [Fetch][Decode][Execute] I3: [Fetch][Decode][Execute] → 5 time slots for 3 instructions → Peak: 1 instruction completes every cycle
PC Offset — Important for Exams

Because of 3-stage pipeline, PC always points 2 instructions ahead of the currently executing instruction.

Executing at 0x1000 → PC = 0x1008 PC = current_address + 8 bytes (4 bytes per ARM instruction × 2 ahead)

GPIO — 3 Steps Every Program
// Step 1: PINSEL — make pin GPIO (00=GPIO)
PINSEL0 &= ~(3 << (2 * pin));
// OR for all: PINSEL0 = 0x00000000;

// Step 2: IO0DIR — set direction
IO0DIR |=  (1 << pin); // output
IO0DIR &= ~(1 << pin); // input

// Step 3: Control
IO0SET = (1 << pin);   // HIGH
IO0CLR = (1 << pin);   // LOW
IO0PIN ^= (1 << pin);  // toggle
IO0PIN & (1 << pin)    // read

// Use #define to name pins:
#define LED (1<<10)   // P0.10
IO0SET = LED;   IO0CLR = LED;

Lab 1 — LED Blinking using 8051
Toggle P1.0 every 500ms using software delay
AssemblyEmbedded C
Circuit + Concept
P1.0 ──[330Ω]──▶|── GND When P1.0=1 → current flows → LED ON When P1.0=0 → no current → LED OFF
Assembly Code
      ORG   0000H
      LJMP  MAIN         ; skip vector table

      ORG   0030H        ; safe area
MAIN:
      MOV   SP, #50H     ; set stack pointer

LOOP:
      SETB  P1.0         ; P1.0 = 1 → LED ON
      LCALL DELAY        ; wait 500ms
      CLR   P1.0         ; P1.0 = 0 → LED OFF
      LCALL DELAY        ; wait 500ms
      SJMP  LOOP         ; repeat forever

; Delay ~500ms at 12MHz
; Inner: 250 x 2us = 500us
; Middle: 250 x 500us = 125ms
; Outer: 4 x 125ms = 500ms
DELAY:
      MOV   R5, #4
OUTER:MOV   R6, #250
MID:  MOV   R7, #250
INNR: DJNZ  R7, INNR
      DJNZ  R6, MID
      DJNZ  R5, OUTER
      RET
      END
Embedded C Code
#include <reg51.h>

sbit LED = P1^0;   // P1.0 = LED pin

void delay_ms(unsigned int ms) {
    unsigned int i, j;
    for(i=0; i<ms; i++)
        for(j=0; j<125; j++);
}

void main() {
    while(1) {
        LED = 1;          // LED ON
        delay_ms(500);
        LED = 0;          // LED OFF
        delay_ms(500);
    }
}

; LCD — Data=P1, RS=P2.0, EN=P2.1
      ORG   0000H
      LJMP  MAIN
      ORG   0030H
MAIN: MOV   SP, #50H
      LCALL LCD_INIT
      MOV   A, #80H     ; line 1
      LCALL LCD_CMD
      MOV   A, #'H'
      LCALL LCD_DATA
      MOV   A, #'I'
      LCALL LCD_DATA
      MOV   A, #0C0H    ; line 2
      LCALL LCD_CMD
      MOV   A, #'8'
      LCALL LCD_DATA
      MOV   A, #'0'
      LCALL LCD_DATA
      MOV   A, #'5'
      LCALL LCD_DATA
      MOV   A, #'1'
      LCALL LCD_DATA
DONE: SJMP  DONE

LCD_INIT:
      MOV A,#38H  &  LCALL LCD_CMD
      MOV A,#0CH  &  LCALL LCD_CMD
      MOV A,#06H  &  LCALL LCD_CMD
      MOV A,#01H  &  LCALL LCD_CMD
      RET

LCD_CMD:                ; A = command byte
      MOV   P1, A       ; put on data bus
      CLR   P2.0        ; RS=0 (command)
      SETB  P2.1        ; EN=1
      LCALL SDLY
      CLR   P2.1        ; EN=0 ← reads here
      LCALL SDLY
      RET

LCD_DATA:               ; A = character
      MOV   P1, A
      SETB  P2.0        ; RS=1 (data)
      SETB  P2.1
      LCALL SDLY
      CLR   P2.1
      LCALL SDLY
      RET

SDLY: MOV R7,#250
S1:   DJNZ R7,S1
      RET
      END
Lab 2 — Embedded C Code
#include <reg51.h>

sbit RS = P2^0;  // Register Select
sbit EN = P2^1;  // Enable
// Data bus = P1 (all 8 bits)

void delay_ms(unsigned int ms){
    unsigned int i,j;
    for(i=0;i<ms;i++) for(j=0;j<125;j++);
}

// Pulse EN to make LCD latch the data
void pulse_en() {
    EN = 1;          // EN HIGH
    delay_ms(1);
    EN = 0;          // EN LOW ← LCD reads here
    delay_ms(2);
}

// Send a COMMAND (RS=0)
void lcd_cmd(unsigned char cmd) {
    P1 = cmd;        // put command on data bus
    RS = 0;          // RS=0 → command mode
    pulse_en();
}

// Send a CHARACTER (RS=1)
void lcd_char(unsigned char ch) {
    P1 = ch;
    RS = 1;          // RS=1 → data mode
    pulse_en();
}

// Print a whole string
void lcd_print(char *str) {
    while(*str)
        lcd_char(*str++);
}

// Initialize LCD — always call first
void lcd_init() {
    lcd_cmd(0x38);   // 8-bit, 2-line
    lcd_cmd(0x0C);   // display ON
    lcd_cmd(0x06);   // cursor right
    lcd_cmd(0x01);   // clear screen
    delay_ms(5);     // clear needs extra time
}

void main() {
    lcd_init();

    lcd_cmd(0x80);           // go to line 1
    lcd_print("HELLO WORLD");

    lcd_cmd(0xC0);           // go to line 2
    lcd_print("8051 EMBEDDED");

    while(1);
}

Lab 3.2 — Serial Transmission (UART)
Send and receive bytes at 9600 baud using Timer1 Mode2
SCON SBUF
Circuit + Key Facts
P3.1 (TXD) ──► MAX232 ──► PC serial port P3.0 (RXD) ◄── MAX232 ◄── PC serial port Setup: TMOD=0x20, TH1=0xFD, SCON=0x50, TR1=1 Crystal: 11.0592 MHz for exact 9600 baud
  • TMOD0x20 = Timer1, Mode2 (8-bit auto-reload)
  • TH10xFD = reload value for 9600 baud @ 11.0592MHz
  • SCON0x50 = Mode1 UART (8-bit), REN=1 (receive enabled)
  • TR1Start Timer1 = baud rate generator
  • TIGoes HIGH when TX complete — MUST clear after
  • RIGoes HIGH when byte received — MUST clear after
⚠If you forget TI=0 or RI=0, the UART locks up on the next byte — flags never get cleared again.
Embedded C Code
#include <reg51.h>

// Initialize UART at 9600 baud
void serial_init() {
    TMOD = 0x20;  // Timer1, Mode2 (auto-reload)
    TH1  = 0xFD;  // 9600 baud @ 11.0592MHz
    SCON = 0x50;  // Mode1 UART, receive enabled
    TR1  = 1;     // start Timer1
}

// Send one character
void send_char(unsigned char ch) {
    SBUF = ch;          // load into TX buffer
    while(TI == 0);    // wait until TX done
    TI   = 0;          // MUST clear TI flag!
}

// Send a string
void send_string(char *str) {
    while(*str)
        send_char(*str++);
}

// Receive one character (blocking)
unsigned char recv_char() {
    unsigned char ch;
    while(RI == 0);    // wait until received
    ch = SBUF;          // read received byte
    RI  = 0;           // MUST clear RI flag!
    return ch;
}

void main() {
    unsigned char received;

    serial_init();

    send_string("Hello from 8051!\r\n");

    while(1) {
        received = recv_char();    // wait for byte
        send_char(received);       // echo it back
    }
}

Lab 4.2 — External Interrupt INT0 (Program 1)
Button on P3.2 triggers INT0 — toggle LED on each press
interrupt 0
Circuit + Concept
Button ── P3.2 (INT0) ── GND LED ── P1.0 ── 330Ω ── GND Button pressed → P3.2 goes LOW (falling edge) → INT0 fires → ISR runs → LED toggles
  • IT0=1Edge triggered — fires on falling edge (HIGH→LOW)
  • IT0=0Level triggered — fires while pin stays LOW
  • EX0=1Enable INT0 interrupt
  • EA=1Master enable — MUST be 1 for any interrupt to fire
Embedded C Code
#include <reg51.h>

sbit LED = P1^0;  // LED output

// INT0 ISR — runs every button press
void int0_isr() interrupt 0 {
    LED = !LED;        // toggle LED
}

void main() {
    IT0 = 1;   // falling edge triggered
    EX0 = 1;   // enable INT0
    EA  = 1;   // global interrupt enable

    while(1);  // ISR handles everything
}

Lab 5.2 — P1.7→P1.0 + 100µs Square Wave on P2.5
Continuously copy bit P1.7 to P1.0 WHILE generating 100µs square wave on P2.5 using Timer0 Mode2. XTAL=11.0592MHz
Timer0 Mode2interrupt 1
Problem Breakdown
Task 1: P1.7 → P1.0 (bit copy, continuously) Task 2: Square wave on P2.5 → Period = 100µs → HIGH for 50µs, LOW for 50µs → Toggle P2.5 every 50µs
Timer Calculation
XTAL = 11.0592 MHz
Machine cycle = 12 / 11.0592MHz = 1.0851µs
Need 50µs delay → count = 50 / 1.0851 = 46
Mode2 preload = 256 − 46 = 210 = 0xD2
TH0 = 0xD2, TL0 = 0xD2
Timer overflows every 46 cycles = ~50µs
ISR toggles P2.5 every 50µs
Result: 50µs HIGH + 50µs LOW = 100µs period ✓
Mode2 (8-bit Auto-Reload)
  • TH0Holds the reload value — loaded into TL0 at each overflow
  • TL0Counts up. On overflow → auto-reloads from TH0. No need to reload manually!
  • AdvMuch more precise than Mode1 (no reload delay)

// Timer0 ISR — fires every ~50µs
// Toggles P2.5 → creates 100µs square wave
void timer0_isr() interrupt 1 {
    // Mode2 auto-reloads TH0 into TL0
    // No need to reload manually
    SQUARE_WAVE = !SQUARE_WAVE; // toggle P2.5
}

void main() {

    // Setup Timer0, Mode2 (8-bit auto-reload)
    // TMOD = 0x02 → Timer0, Mode2
    TMOD = 0x02;

    // Preload = 256 - 46 = 210 = 0xD2
    // Both TH0 and TL0 get the same value
    TH0  = 0xD2;  // reload value
    TL0  = 0xD2;  // starting value

    // Enable Timer0 interrupt
    ET0  = 1;     // enable Timer0 interrupt
    EA   = 1;     // global interrupt enable
    TR0  = 1;     // start Timer0

    // Main loop: continuously copy P1.7 to P1.0
    // Timer ISR handles square wave in background
    while(1) {
        OUTPUT_BIT = INPUT_BIT; // copy bit P1.7→P1.0
    }
}

Key GPIO Rules
// IO0SET: writing 1 drives pin HIGH
//         writing 0 has NO effect
// IO0CLR: writing 1 drives pin LOW
//         writing 0 has NO effect

// Set pin HIGH:
IO0SET = (1<<pin);

// Set pin LOW:
IO0CLR = (1<<pin);

// Toggle:
IO0PIN ^= (1<<pin);

// Read pin:
if(IO0PIN & (1<<pin)) { /* HIGH */ }

IO0DIR = 0xFFFFFFFF;        // ALL output
IO0SET = 0xFFFFFFFF;        // ALL HIGH

Embedded C Code
#include <lpc214x.h>

#define COILS  0x0F    // P0.0-P0.3

void delay_ms(int ms){
    int i,j;
    for(i=0;i<ms;i++) for(j=0;j<6000;j++);
}

// Full step sequence — 4 patterns
unsigned char step_seq[4] = {
    0x09, // step 1: coil A+D
    0x03, // step 2: coil A+B
    0x06, // step 3: coil B+C
    0x0C  // step 4: coil C+D
};

// Rotate CLOCKWISE — n steps, speed in ms
void clockwise(int n, int speed) {
    int i;
    for(i=0; i<n; i++) {
        IO0CLR = COILS;          // all coils off
        IO0SET = step_seq[i%4];  // next step
        delay_ms(speed);
    }
}

// Rotate ANTICLOCKWISE — reverse sequence
void anticlockwise(int n, int speed) {
    int i;
    for(i=n-1; i>=0; i--) {
        IO0CLR = COILS;
        IO0SET = step_seq[i%4];
        delay_ms(speed);
    }
}

// Stop motor
void motor_stop() {
    IO0CLR = COILS;             // all coils off
}

int main() {
    PINSEL0 = 0x00000000;
    IO0DIR  = COILS;   // P0.0-P0.3 = output

    while(1) {
        clockwise(200, 5);      // 1 revolution CW
        delay_ms(1000);
        anticlockwise(200, 5);  // 1 revolution CCW
        delay_ms(1000);
    }
}

Circuit + Key Registers
LPC2148 P0.0 (TXD0) ──► USB-UART converter ──► PC LPC2148 P0.1 (RXD0) ◄── USB-UART converter ◄── PC OR via onboard USB if board supports it
RegisterPurpose
U0LCRLine control: word length + DLAB bit
U0DLLBaud rate divisor low (when DLAB=1)
U0DLMBaud rate divisor high (when DLAB=1)
U0LSRStatus: bit5=TX ready, bit0=RX data ready
U0THRWrite here to transmit
U0RBRRead here to receive
💡DLAB=1 lets you access DLL/DLM for baud rate. Always clear DLAB (set to 0) before using UART normally.
Baud divisor = PCLK / (16 × baud)
For 9600, PCLK=15MHz:
divisor = 15000000 / (16×9600) = 97.65 ≈ 98 = 0x62
U0DLM=0x00, U0DLL=0x62
Embedded C Code
#include <lpc214x.h>

// Initialize UART0 at 9600 baud (PCLK=15MHz)
void uart_init() {
    // P0.0=TXD0, P0.1=RXD0
    PINSEL0 |= 0x00000005;

    // DLAB=1 to set baud rate
    U0LCR = 0x83;    // 8-bit, no parity, DLAB=1
    U0DLM = 0x00;    // high byte of divisor
    U0DLL = 0x62;    // low byte (9600 baud)

    // DLAB=0 for normal UART use
    U0LCR = 0x03;    // 8-bit, no parity, DLAB=0
}

// Send one character
void send_char(unsigned char ch) {
    // Wait until TX buffer empty (bit5 of LSR)
    while(!(U0LSR & 0x20));
    U0THR = ch;      // transmit
}

// Send a string
void send_string(char *str) {
    while(*str) send_char(*str++);
}

// Receive one character (blocking)
unsigned char recv_char() {
    // Wait until data ready (bit0 of LSR)
    while(!(U0LSR & 0x01));
    return U0RBR;    // read received byte
}

void delay_ms(int ms){int i,j;for(i=0;i<ms;i++)for(j=0;j<6000;j++);}

int main() {
    unsigned char received;

    uart_init();

    send_string("LPC2148 Ready!\r\n");

    while(1) {
        // Send message every 2 seconds
        send_string("Hello from ARM!\r\n");
        delay_ms(2000);

        // Also echo received bytes back
        if(U0LSR & 0x01) {           // any data?
            received = U0RBR;
            send_char(received);      // echo back
        }
    }
}

#include <reg51.h>

sbit RS = P3^0;
sbit E = P3^1;

sbit LED = P3^7;
sbit LED2 = P1^7;

#define LCD_DATA P2

void delay_old(int ms) {
  int i, j;
  for (i = 0; i < ms; i++)
    for (j = 0; j < 127; j++);
}

void interrupt0ISR() interrupt 0 {
  LED2 ^= 1;
}

void interrupt2ISR() interrupt 2 {
  LED2 ^= 1;
  delay(1000);
}


void interrupt_init() {
  IT0 = 1; // only from high to low
  IT1 = 0; //low level continuous trigger on 
  EX0 = 1;
  EX1 = 1;
  EA = 1;
}

void lcd_pulse() {
  delay(2);
  E = 1;
  delay(2);
  E = 0;
}

void lcd_cmd(unsigned char cmd) {
  RS = 0;
  LCD_DATA = cmd;
  lcd_pulse();
}

void lcd_byte(unsigned char ch) {
  RS = 1;
  LCD_DATA = ch;
  lcd_pulse();
}

void lcd_clr() {
  lcd_cmd(0x01);
}

void lcd_init() {
  RS = 0;
  lcd_cmd(0x38);
  lcd_cmd(0x0E);
  lcd_clr();
}

void lcd_print(char *str) {
  while (*str != 0)
    lcd_byte(*str++);
}

void timer_delay(int ms){
    TMOD = 0x01;   // Timer0 Mode1
    
    while (ms--){
        TH0 = 0xFC;   // Reload each time
        TL0 = 0x66;

        TR0 = 1;      // Start timer
        
        while(TF0 == 0);  // Wait for overflow
        
        TR0 = 0;      // Stop timer
        TF0 = 0;      // Clear flag
    }
}

int main() {
  int i;
  interrupt_init();
  lcd_init();
  lcd_print("JASSI IS GAY");
  for (;;) {
    LED = 1;
    delay(1000);
    LED = 0;
    delay(1000);
  }
}

    for (j = 0; j < 127; j++);
}

void UART_init() {
  TMOD = 0x20;
  TH1 = 0xFD;
  TR1 = 1;
  SCON = 0x50;
}

void UART_transmit(unsigned char ch) {
  SBUF = ch;
  while (TI == 0);
  TI = 0;
}

unsigned char UART_receive() {
  while (RI == 0);
  RI = 0;
  return SBUF;
}

void UART_string(char *str) {
  while (*str != 0)
    UART_transmit(*str++);
}

#include <lpc214x.h>

#define RS (1 << 16)
#define RW (1 << 17)
#define E (1 << 18)

void LCD_pulse() {
  IOSET0 = E;
  delay(2);
  IOCLR0 = E;
  delay(2);
}

void LCD_cmd(char ch) {
  IOCLR0 = RS;
  IOCLR1 = 0xFF << 16;
  IOSET1 = (int)ch << 16;
	
  LCD_pulse();
}

void LCD_data(char ch) {
  IOSET0 = RS;
  IOCLR1 = 0xFF << 16;
  IOSET1 = (int)ch << 16;
	
  LCD_pulse();
}

void LCD_str(char *str) {
  while (*str != 0)
    LCD_data(*str++);
}

void LCD_init() {
  IODIR0 |= 7 << 16;
  IODIR1 |= 0xFF << 16;
  LCD_cmd(0x38);
  LCD_cmd(0x0E);
}

// serial

#include <lpc214x.h>

void UART_init() {
  VPBDIV = 1;
  PINSEL0 |= 5; // 0101
  U0LCR = (1 << 7);
  U0DLL = 65;
  U0DLM = 0;
  U0LCR = 3;
}

void UART_send(char ch) {
  while (!(U0LSR & (1 << 5)));
  U0THR = ch;
}

void UART_send_str(char *str) {
  while (*str != 0)
    UART_send(*str++);
}

char UART_recv() {
  while (!(U0LSR & 1));
  return U0RBR;
}

int main() {
  UART_init();
  UART_send_str("hello world");
  while (1);
}
